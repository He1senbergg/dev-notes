```
#!/bin/bash

while true; do
    # 打印当前时间
	echo "---------------------------------------------"
    echo "Current time: $(date '+%Y-%m-%d %H:%M:%S')"

    echo "LLM"
    cat /proc/132/status | grep VmRSS
    cat /proc/132/status | grep VmSize

    # 等待 10 分钟
    sleep 60
done
```

用这个脚本把内存打印出来，132换成自己进程的pid
