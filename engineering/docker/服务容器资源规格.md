测定服务容器的<span style="color:#e60000;">**平均CPU、平均MEM**</span>资源限额。

```
cat > collect_container_resource_auto.sh <<'EOF'
#!/bin/bash

interval=${1:-1}
out=${2:-container_resource_$(date +%Y%m%d_%H%M%S).csv}

CGROUP_MODE="unknown"

CPU_USAGE_MODE=""
CPU_USAGE_FILE=""
CPU_QUOTA_FILE=""
CPU_PERIOD_FILE=""
CPU_MAX_FILE=""
CPU_LIMIT_CORE="0"

MEM_USAGE_FILE=""
MEM_LIMIT_FILE=""
MEM_STAT_FILE=""

if [ -f /sys/fs/cgroup/cgroup.controllers ]; then
    CGROUP_MODE="v2"
else
    CGROUP_MODE="v1"
fi

if [ "$CGROUP_MODE" = "v2" ]; then
    CPU_USAGE_MODE="v2_usage_usec"
    CPU_USAGE_FILE="/sys/fs/cgroup/cpu.stat"
    CPU_MAX_FILE="/sys/fs/cgroup/cpu.max"

    if [ ! -f "$CPU_USAGE_FILE" ]; then
        echo "[collect_container_resource_auto] cpu.stat not found: $CPU_USAGE_FILE"
        exit 1
    fi

    if [ -f "$CPU_MAX_FILE" ]; then
        cpu_max_quota=$(awk '{print $1}' "$CPU_MAX_FILE")
        cpu_max_period=$(awk '{print $2}' "$CPU_MAX_FILE")

        if [ "$cpu_max_quota" = "max" ]; then
            CPU_LIMIT_CORE="0"
        else
            CPU_LIMIT_CORE=$(awk -v q="$cpu_max_quota" -v p="$cpu_max_period" 'BEGIN {printf "%.6f", q/p}')
        fi
    fi

    MEM_USAGE_FILE="/sys/fs/cgroup/memory.current"
    MEM_LIMIT_FILE="/sys/fs/cgroup/memory.max"
    MEM_STAT_FILE="/sys/fs/cgroup/memory.stat"

else
    CPU_CG=$(awk -F: '$2=="cpu,cpuacct"{print $3}' /proc/self/cgroup)
    MEM_CG=$(awk -F: '$2=="memory"{print $3}' /proc/self/cgroup)

    CPU_USAGE_MODE="v1_usage_ns"

    for f in \
        "/sys/fs/cgroup/cpuacct${CPU_CG}/cpuacct.usage" \
        "/sys/fs/cgroup/cpu,cpuacct${CPU_CG}/cpuacct.usage" \
        "/sys/fs/cgroup/cpuacct/cpuacct.usage" \
        "/sys/fs/cgroup/cpu,cpuacct/cpuacct.usage" \
        "/sys/fs/cgroup/cpu/cpuacct.usage"
    do
        if [ -f "$f" ]; then
            CPU_USAGE_FILE="$f"
            break
        fi
    done

    for f in \
        "/sys/fs/cgroup/cpu${CPU_CG}/cpu.cfs_quota_us" \
        "/sys/fs/cgroup/cpu,cpuacct${CPU_CG}/cpu.cfs_quota_us" \
        "/sys/fs/cgroup/cpu/cpu.cfs_quota_us" \
        "/sys/fs/cgroup/cpu,cpuacct/cpu.cfs_quota_us"
    do
        if [ -f "$f" ]; then
            CPU_QUOTA_FILE="$f"
            CPU_PERIOD_FILE="${f%cpu.cfs_quota_us}cpu.cfs_period_us"
            break
        fi
    done

    if [ -z "$CPU_USAGE_FILE" ]; then
        echo "[collect_container_resource_auto] cpu usage file not found"
        echo "[collect_container_resource_auto] CPU_CG=$CPU_CG"
        exit 1
    fi

    if [ -n "$CPU_QUOTA_FILE" ] && [ -f "$CPU_QUOTA_FILE" ]; then
        quota=$(cat "$CPU_QUOTA_FILE")
        period=$(cat "$CPU_PERIOD_FILE")
        if [ "$quota" -gt 0 ]; then
            CPU_LIMIT_CORE=$(awk -v q="$quota" -v p="$period" 'BEGIN {printf "%.6f", q/p}')
        fi
    fi

    if [ -f "/sys/fs/cgroup/memory${MEM_CG}/memory.usage_in_bytes" ]; then
        MEM_USAGE_FILE="/sys/fs/cgroup/memory${MEM_CG}/memory.usage_in_bytes"
        MEM_LIMIT_FILE="/sys/fs/cgroup/memory${MEM_CG}/memory.limit_in_bytes"
        MEM_STAT_FILE="/sys/fs/cgroup/memory${MEM_CG}/memory.stat"
    elif [ -f "/sys/fs/cgroup/memory/memory.usage_in_bytes" ]; then
        MEM_USAGE_FILE="/sys/fs/cgroup/memory/memory.usage_in_bytes"
        MEM_LIMIT_FILE="/sys/fs/cgroup/memory/memory.limit_in_bytes"
        MEM_STAT_FILE="/sys/fs/cgroup/memory/memory.stat"
    else
        echo "[collect_container_resource_auto] memory usage file not found"
        echo "[collect_container_resource_auto] MEM_CG=$MEM_CG"
        exit 1
    fi
fi

if [ ! -f "$MEM_USAGE_FILE" ]; then
    echo "[collect_container_resource_auto] memory usage file not found: $MEM_USAGE_FILE"
    exit 1
fi

if [ ! -f "$MEM_LIMIT_FILE" ]; then
    echo "[collect_container_resource_auto] memory limit file not found: $MEM_LIMIT_FILE"
    exit 1
fi

echo "[collect_container_resource_auto] interval=${interval}s"
echo "[collect_container_resource_auto] output=$out"
echo "[collect_container_resource_auto] CGROUP_MODE=$CGROUP_MODE"
echo "[collect_container_resource_auto] CPU_USAGE_MODE=$CPU_USAGE_MODE"
echo "[collect_container_resource_auto] CPU_USAGE_FILE=$CPU_USAGE_FILE"
echo "[collect_container_resource_auto] CPU_LIMIT_CORE=$CPU_LIMIT_CORE"
echo "[collect_container_resource_auto] MEM_USAGE_FILE=$MEM_USAGE_FILE"
echo "[collect_container_resource_auto] MEM_LIMIT_FILE=$MEM_LIMIT_FILE"
echo "[collect_container_resource_auto] MEM_STAT_FILE=$MEM_STAT_FILE"

echo "time,cpu_core,cpu_percent_of_limit,mem_usage_bytes,mem_usage_mib,mem_percent_of_limit,mem_limit_bytes,mem_limit_mib,total_process_rss_mib" > "$out"

read_cpu_usage_ns() {
    if [ "$CPU_USAGE_MODE" = "v2_usage_usec" ]; then
        usage_usec=$(awk '$1=="usage_usec" {print $2}' "$CPU_USAGE_FILE")
        echo $((usage_usec * 1000))
    else
        cat "$CPU_USAGE_FILE"
    fi
}

read_mem_limit_bytes() {
    value=$(cat "$MEM_LIMIT_FILE")
    if [ "$value" = "max" ]; then
        echo 0
    else
        echo "$value"
    fi
}

prev_cpu=$(read_cpu_usage_ns)
prev_time=$(date +%s%N)

while true; do
    sleep "$interval"

    now=$(date '+%F %T')

    curr_cpu=$(read_cpu_usage_ns)
    curr_time=$(date +%s%N)

    delta_cpu=$((curr_cpu - prev_cpu))
    delta_time=$((curr_time - prev_time))

    mem_usage=$(cat "$MEM_USAGE_FILE")
    mem_limit=$(read_mem_limit_bytes)

    total_process_rss_kb=$(ps -eo rss= | awk '{sum += $1} END {print sum+0}')

    awk -v t="$now" -v dc="$delta_cpu" -v dt="$delta_time" -v cpu_limit="$CPU_LIMIT_CORE" -v mu="$mem_usage" -v ml="$mem_limit" -v prss="$total_process_rss_kb" 'BEGIN {
        cpu_core = dc / dt;

        if (cpu_limit > 0) {
            cpu_percent_limit = cpu_core / cpu_limit * 100;
        } else {
            cpu_percent_limit = 0;
        }

        if (ml > 0) {
            mem_percent_limit = mu / ml * 100;
            mem_limit_mib = ml / 1024 / 1024;
        } else {
            mem_percent_limit = 0;
            mem_limit_mib = 0;
        }

        printf "%s,%.6f,%.2f,%s,%.2f,%.2f,%s,%.2f,%.2f\n", t, cpu_core, cpu_percent_limit, mu, mu/1024/1024, mem_percent_limit, ml, mem_limit_mib, prss/1024;
        }'

    prev_cpu=$curr_cpu
    prev_time=$curr_time
done >> "$out"
EOF
```
